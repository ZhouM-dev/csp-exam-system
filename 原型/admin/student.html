23              for (int coupons = 0; coupons <= n; ++coupons) {
24                  ll cost = dp[mask][coupons];

26                  for (int i = 0; i < n; ++i) {
27                      if (mask & (1 << i)) continue;
28                      int next = mask | (1 << i);
29                      if (coupons < n)
30                          dp[next][coupons + 1] = min(dp[next][coupons + 1], cost + a[i]);
